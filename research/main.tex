% Cross-reference navigation. hyperref makes every \cite, \ref, and \url a live
% link (citation -> bibliography entry, "Figure 4" -> the figure, and so on);
% backref=page adds the return path, printing "(cited on page N)" after each
% bibliography entry and linking back to where it was used. Both directions
% matter for a document read on screen rather than on paper.
%
% Loaded LAST, deliberately: hyperref redefines a large number of internals and
% expects to see the final versions of everything it patches, so a package
% loaded after it can silently break the links.
%
% Links are coloured rather than boxed, in the same xcolor `blue` the fox
% figures shade their cells with, so a link reads as part of the document's
% palette instead of as browser chrome.
\usepackage[colorlinks=true,allcolors=blue,backref=page]{hyperref}

% Reverse links WITHOUT a "(cited on page 5)" trailer, which would be an
% editorial addition to every bibliography entry and is not part of the
% reference format. Instead the entry's own numeric label - the [5] - becomes
% the link back to where the work was first cited. Nothing is added to the
% page; an existing element is made live.
%
% This works because of when backref runs. It writes \backcite{key}{{page}{sec}
% {anchor}} into main.brf, and that file is read at \begin{document} - before
% any \bibitem is typeset - so by the time a label needs printing, the anchor
% for its key is already known.
\makeatletter
% Record the FIRST citing anchor per key (later ones are discarded; a label is
% one link, and the first use is the one a reader wants).
%
% #1 is NOT always a single key: backref records a multi-key \cite{a,b} under
% the joined string "a,b", which matches neither key. Left unsplit, every key in
% a grouped citation silently loses its backlink - that was 12 of this paper's
% entries. So iterate the comma list. The \@firstofone line is the LaTeX
% kernel's own idiom (\@citex uses it on exactly this kind of list) for dropping
% the leading space in "\cite{a, b}"; \zap@space cannot do this job because it
% needs a real space token and we are inside \edef.
\renewcommand*{\backcite}[2]{\bkref@keys{#1}#2}
\def\bkref@keys#1#2#3#4{%
  \@for\bkref@k:=#1\do{%
    \edef\bkref@k{\expandafter\@firstofone\bkref@k\@empty}%
    \@ifundefined{bkref@\bkref@k}%
      {\expandafter\xdef\csname bkref@\bkref@k\endcsname{#4}}%
      {}}}
% Suppress backref's own trailing text entirely.
\renewcommand*{\backref}[1]{}
\renewcommand*{\backrefalt}[4]{}
% \@biblabel receives only the number, not the key, so stash the key that
% \bibitem is currently emitting and let the label look up its anchor.
\gdef\bkref@curkey{}
\let\bkref@orig@bibitem\@bibitem
\def\@bibitem#1{\gdef\bkref@curkey{#1}\bkref@orig@bibitem{#1}}
\renewcommand{\@biblabel}[1]{%
  \@ifundefined{bkref@\bkref@curkey}%
    {[#1]}%
    {\hyperlink{\csname bkref@\bkref@curkey\endcsname}{[#1]}}}
\makeatother
